create database uber;

use uber;

create table users (
  id           int  primary key,
  full_name     varchar(20) not null,
  email         varchar(30) unique,
  gender        char(1)    not null check(gender = 'm' or gender = 'f'),
  date_of_birth  varchar(15) not null,
  created_at    datetime,
  country_code  int
);

create table drivers (
  id  int  primary key,
  full_name varchar(20) not null,
  license_number  varchar(20) unique not null,
  rating decimal(3,2) default 5.00,
  is_available boolean  default true,
  created_at datetime
);

create table trips (
  id int primary key,
  user_id int not null,
  driver_id  int not null,
  distance_km  decimal(6,2) not null,
  status varchar(10)  check(status = 'pending' or status = 'completed' or status = 'cancelled'),
  created_at  datetime,
  foreign key (user_id) references users(id),
  foreign key (driver_id) references drivers(id)
);

create table payments (
  id int  primary key,
  trip_id int  not null unique,
  amount  decimal(8,2) not null,
  method  varchar(20)  check(method = 'cash' or method = 'card' or method = 'applePay'),
  status  varchar(10)  check(status = 'pending' or status = 'paid' or status = 'refunded'),
  created_at datetime,
  foreign key (trip_id) references trips(id)
);

alter table users    modify created_at datetime default current_timestamp;
alter table drivers  modify created_at datetime default current_timestamp;
alter table trips    modify created_at datetime default current_timestamp;
alter table payments modify created_at datetime default current_timestamp;
